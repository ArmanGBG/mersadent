     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Card, CardContent } from '@/components/ui/card';
     9→import { Skeleton } from '@/components/ui/skeleton';
    10→import { Switch } from '@/components/ui/switch';
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    12→import { Textarea } from '@/components/ui/textarea';
    13→import { Separator } from '@/components/ui/separator';
    14→import {
    15→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription,
    16→} from '@/components/ui/dialog';
    17→import {
    18→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    19→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    20→} from '@/components/ui/alert-dialog';
    21→import {
    22→  Search, Plus, Pencil, Trash2, Upload, X, ChevronLeft, ChevronRight, ImageIcon, Loader2, Wand2, FileText, CheckCircle2, AlertTriangle
    23→} from 'lucide-react';
    24→import { toast } from 'sonner';
    25→import { adminFetch, formatNumber, toSlug } from '@/lib/admin-utils';
    26→import { parseSuperPrompt } from '@/lib/parse-super-prompt';
    27→
    28→interface AdminProduct {
    29→  id: number; name: string; slug: string; model: string | null;
    30→  price: number | null; priceType: string; availability: string;
    31→  specs: string | null; featured: boolean; visible: boolean; order: number; discountPercent: number;
    32→  description: string | null;
    33→  strengths: string | null;
    34→  weaknesses: string | null;
    35→  metaTitle: string | null;
    36→  metaDescription: string | null;
    37→  keywords: string | null;
    38→  sku: string | null;
    39→  country: string | null;
    40→  warranty: string | null;
    41→  weight: string | null;
    42→  lowStock: number;
    43→  category: { id: number; name: string; slug: string };
    44→  brand: { id: number; name: string; slug: string };
    45→  _count: { images: number };
    46→}
    47→
    48→interface SimpleCategory { id: number; name: string; slug: string; parentId: number | null; }
    49→interface SimpleBrand { id: number; name: string; slug: string; }
    50→
    51→interface ProductForm {
    52→  name: string; slug: string; model: string; price: string; priceType: string;
    53→  availability: string; categoryId: string; brandId: string;
    54→  featured: boolean; visible: boolean; discountPercent: string;
    55→  description: string; strengths: string[]; weaknesses: string[];
    56→  metaTitle: string; metaDescription: string; keywords: string;
    57→  sku: string; country: string; warranty: string; weight: string; lowStock: string;
    58→}
    59→
    60→const emptyProduct: ProductForm = {
    61→  name: '', slug: '', model: '', price: '', priceType: 'fixed', availability: 'in_stock',
    62→  categoryId: '', brandId: '', featured: false, visible: true, discountPercent: '0',
    63→  description: '', strengths: [], weaknesses: [], metaTitle: '', metaDescription: '', keywords: '',
    64→  sku: '', country: '', warranty: '', weight: '', lowStock: '0',
    65→};
    66→
    67→export function AdminProducts({ token }: { token: string }) {
    68→  const [products, setProducts] = useState<AdminProduct[]>([]);
    69→  const [total, setTotal] = useState(0);
    70→  const [page, setPage] = useState(1);
    71→  const [q, setQ] = useState('');
    72→  const [loading, setLoading] = useState(true);
    73→  const [searchTimeout, setSearchTimeout] = useState<NodeJS.Timeout | null>(null);
    74→
    75→  const [categories, setCategories] = useState<SimpleCategory[]>([]);
    76→  const [brands, setBrands] = useState<SimpleBrand[]>([]);
    77→
    78→  // Dialog state
    79→  const [dialogOpen, setDialogOpen] = useState(false);
    80→  const [editing, setEditing] = useState<AdminProduct | null>(null);
    81→  const [form, setForm] = useState<ProductForm>({ ...emptyProduct });
    82→  const [saving, setSaving] = useState(false);
    83→
    84→  // Super-prompt state
    85→  const [superPromptText, setSuperPromptText] = useState('');
    86→  const [parsing, setParsing] = useState(false);
    87→
    88→  // Delete state
    89→  const [deleteTarget, setDeleteTarget] = useState<AdminProduct | null>(null);
    90→  const [deleting, setDeleting] = useState(false);
    91→
    92→  // Image dialog
    93→  const [imageProduct, setImageProduct] = useState<AdminProduct | null>(null);
    94→  const [images, setImages] = useState<{ id: number; url: string; alt: string | null; order: number }[]>([]);
    95→  const [uploading, setUploading] = useState(false);
    96→
    97→  const limit = 20;
    98→
    99→  const loadProducts = useCallback(async () => {
   100→    setLoading(true);
   101→    try {
   102→      const res = await adminFetch(token, `/api/admin/products?page=${page}&limit=${limit}&q=${encodeURIComponent(q)}`);
   103→      const data = await res.json();
   104→      if (data.error) { toast.error(data.error); return; }
   105→      setProducts(data.products);
   106→      setTotal(data.total);
   107→    } catch { toast.error('خطا در دریافت محصولات'); }
   108→    finally { setLoading(false); }
   109→  }, [token, page, q]);
   110→
   111→  const loadLookups = useCallback(async () => {
   112→    try {
   113→      const [catRes, brandRes] = await Promise.all([
   114→        adminFetch(token, '/api/admin/categories?flat=true'),
   115→        adminFetch(token, '/api/admin/brands'),
   116→      ]);
   117→      const catData = await catRes.json();
   118→      const brandData = await brandRes.json();
   119→      if (catData.categories) setCategories(catData.categories);
   120→      if (brandData.brands) setBrands(brandData.brands);
   121→    } catch { /* ignore */ }
   122→  }, [token]);
   123→
   124→  useEffect(() => { loadLookups(); }, [loadLookups]);
   125→  useEffect(() => { loadProducts(); }, [loadProducts]);
   126→
   127→  const handleSearch = (value: string) => {
   128→    setQ(value);
   129→    setPage(1);
   130→    if (searchTimeout) clearTimeout(searchTimeout);
   131→    setSearchTimeout(setTimeout(() => {}, 100));
   132→  };
   133→
   134→  const openCreate = () => {
   135→    setEditing(null);
   136→    setForm({ ...emptyProduct });
   137→    setSuperPromptText('');
   138→    setDialogOpen(true);
   139→  };
   140→
   141→  const openEdit = (p: AdminProduct) => {
   142→    setEditing(p);
   143→    setForm({
   144→      name: p.name,
   145→      slug: p.slug,
   146→      model: p.model || '',
   147→      price: p.price !== null ? String(p.price) : '',
   148→      priceType: p.priceType,
   149→      availability: p.availability,
   150→      categoryId: String(p.category.id),
   151→      brandId: String(p.brand.id),
   152→      featured: p.featured,
   153→      visible: p.visible !== false,
   154→      discountPercent: String(p.discountPercent ?? 0),
   155→      description: p.description || '',
   156→      strengths: p.strengths ? (() => { try { return JSON.parse(p.strengths); } catch { return []; } })() : [],
   157→      weaknesses: p.weaknesses ? (() => { try { return JSON.parse(p.weaknesses); } catch { return []; } })() : [],
   158→      metaTitle: p.metaTitle || '',
   159→      metaDescription: p.metaDescription || '',
   160→      keywords: p.keywords || '',
   161→      sku: p.sku || '',
   162→      country: p.country || '',
   163→      warranty: p.warranty || '',
   164→      weight: p.weight || '',
   165→      lowStock: String(p.lowStock ?? 0),
   166→    });
   167→    setSuperPromptText('');
   168→    setDialogOpen(true);
   169→  };
   170→
   171→  const handleSave = async () => {
   172→    if (!form.name || !form.categoryId || !form.brandId) {
   173→      toast.error('نام، دسته‌بندی و برند الزامی هستند');
   174→      return;
   175→    }
   176→    setSaving(true);
   177→    try {
   178→      const body: Record<string, unknown> = {
   179→        name: form.name,
   180→        slug: form.slug || toSlug(form.name),
   181→        model: form.model || null,
   182→        price: form.price === '' ? null : Number(form.price),
   183→        discountPercent: Number(form.discountPercent) || 0,
   184→        priceType: form.priceType,
   185→        availability: form.availability,
   186→        categoryId: Number(form.categoryId),
   187→        brandId: Number(form.brandId),
   188→        featured: form.featured,
   189→        visible: form.visible,
   190→        description: form.description || null,
   191→        strengths: form.strengths.length > 0 ? JSON.stringify(form.strengths) : null,
   192→        weaknesses: form.weaknesses.length > 0 ? JSON.stringify(form.weaknesses) : null,
   193→        metaTitle: form.metaTitle || null,
   194→        metaDescription: form.metaDescription || null,
   195→        keywords: form.keywords || null,
   196→        sku: form.sku || null,
   197→        country: form.country || null,
   198→        warranty: form.warranty || null,
   199→        weight: form.weight || null,
   200→        lowStock: form.lowStock || '0',
   201→      };
   202→
   203→      if (editing) {
   204→        body.id = editing.id;
   205→        const res = await adminFetch(token, '/api/admin/products', { method: 'PUT', body: JSON.stringify(body) });
   206→        const data = await res.json();
   207→        if (data.error) { toast.error(data.error); return; }
   208→        toast.success('محصول بروزرسانی شد');
   209→      } else {
   210→        const res = await adminFetch(token, '/api/admin/products', { method: 'POST', body: JSON.stringify(body) });
   211→        const data = await res.json();
   212→        if (data.error) { toast.error(data.error); return; }
   213→        toast.success('محصول ایجاد شد');
   214→      }
   215→      setDialogOpen(false);
   216→      loadProducts();
   217→    } catch { toast.error('خطا در ذخیره محصول'); }
   218→    finally { setSaving(false); }
   219→  };
   220→
   221→  const handleParseSuperPrompt = () => {
   222→    if (!superPromptText.trim()) {
   223→      toast.error('ابتدا متن خروجی را وارد کنید');
   224→      return;
   225→    }
   226→    setParsing(true);
   227→    // Use setTimeout to allow UI to update before synchronous parse
   228→    setTimeout(() => {
   229→      try {
   230→        const parsed = parseSuperPrompt(superPromptText);
   231→        setForm(f => ({
   232→          ...f,
   233→          description: parsed.description || f.description,
   234→          strengths: parsed.strengths.length > 0 ? parsed.strengths : f.strengths,
   235→          weaknesses: parsed.weaknesses.length > 0 ? parsed.weaknesses : f.weaknesses,
   236→          metaTitle: parsed.metaTitle || f.metaTitle,
   237→          metaDescription: parsed.metaDescription || f.metaDescription,
   238→          keywords: parsed.keywords || f.keywords,
   239→        }));
   240→        const foundCount = [
   241→          parsed.description ? 'توضیحات' : '',
   242→          parsed.strengths.length ? `${parsed.strengths.length} نقطه قوت` : '',
   243→          parsed.weaknesses.length ? `${parsed.weaknesses.length} نقطه ضعف` : '',
   244→        ].filter(Boolean);
   245→        if (foundCount.length > 0) {
   246→          toast.success(`تحلیل موفق: ${foundCount.join('، ')}`);
   247→        } else {
   248→          toast.warning('متن پردازش شد اما بخش مشخصی یافت نشد');
   249→        }
   250→      } catch (e) {
   251→        toast.error('خطا در تحلیل متن');
   252→        console.error('Parse error:', e);
   253→      } finally {
   254→        setParsing(false);
   255→      }
   256→    }, 50);
   257→  };
   258→
   259→  const removeStrength = (idx: number) => {
   260→    setForm(f => ({ ...f, strengths: f.strengths.filter((_, i) => i !== idx) }));
   261→  };
   262→
   263→  const removeWeakness = (idx: number) => {
   264→    setForm(f => ({ ...f, weaknesses: f.weaknesses.filter((_, i) => i !== idx) }));
   265→  };
   266→
   267→  const handleDelete = async () => {
   268→    if (!deleteTarget) return;
   269→    setDeleting(true);
   270→    try {
   271→      const res = await adminFetch(token, `/api/admin/products?id=${deleteTarget.id}`, { method: 'DELETE' });
   272→      const data = await res.json();
   273→      if (data.error) { toast.error(data.error); return; }
   274→      toast.success('محصول حذف شد');
   275→      setDeleteTarget(null);
   276→      loadProducts();
   277→    } catch { toast.error('خطا در حذف محصول'); }
   278→    finally { setDeleting(false); }
   279→  };
   280→
   281→  const openImages = async (p: AdminProduct) => {
   282→    setImageProduct(p);
   283→    try {
   284→      const res = await adminFetch(token, `/api/admin/products/${p.id}/images`);
   285→      const data = await res.json();
   286→      setImages(data.images || []);
   287→    } catch { setImages([]); }
   288→  };
   289→
   290→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   291→    const file = e.target.files?.[0];
   292→    if (!file || !imageProduct) return;
   293→    setUploading(true);
   294→    try {
   295→      const fd = new FormData();
   296→      fd.append('file', file);
   297→      fd.append('productId', String(imageProduct.id));
   298→      const res = await fetch('/api/admin/upload', { method: 'POST', body: fd });
   299→      const data = await res.json();
   300→      if (data.error) { toast.error(data.error); return; }
   301→      toast.success('تصویر آپلود شد');
   302→      openImages(imageProduct);
   303→    } catch { toast.error('خطا در آپلود'); }
   304→    finally { setUploading(false); e.target.value = ''; }
   305→  };
   306→
   307→  const handleDeleteImage = async (imageId: number) => {
   308→    if (!imageProduct) return;
   309→    try {
   310→      const res = await adminFetch(token, `/api/admin/products/${imageProduct.id}/images?imageId=${imageId}`, { method: 'DELETE' });
   311→      const data = await res.json();
   312→      if (data.error) { toast.error(data.error); return; }
   313→      toast.success('تصویر حذف شد');
   314→      setImages(prev => prev.filter(i => i.id !== imageId));
   315→    } catch { toast.error('خطا در حذف تصویر'); }
   316→  };
   317→
   318→  const toggleFeatured = async (p: AdminProduct) => {
   319→    try {
   320→      const res = await adminFetch(token, '/api/admin/products', {
   321→        method: 'PUT',
   322→        body: JSON.stringify({ id: p.id, name: p.name, featured: !p.featured }),
   323→      });
   324→      const data = await res.json();
   325→      if (data.error) { toast.error(data.error); return; }
   326→      toast.success(p.featured ? 'از ویژه‌ها حذف شد' : 'به ویژه‌ها اضافه شد');
   327→      loadProducts();
   328→    } catch { toast.error('خطا'); }
   329→  };
   330→
   331→  const toggleVisible = async (p: AdminProduct) => {
   332→    const newVal = !(p.visible !== false);
   333→    // Optimistic
   334→    setProducts(prev => prev.map(x => x.id === p.id ? { ...x, visible: newVal } : x));
   335→    try {
   336→      const res = await adminFetch(token, '/api/admin/products', {
   337→        method: 'PUT',
   338→        body: JSON.stringify({ id: p.id, name: p.name, visible: newVal }),
   339→      });
   340→      const data = await res.json();
   341→      if (data.error) {
   342→        toast.error(data.error);
   343→        loadProducts();
   344→        return;
   345→      }
   346→      toast.success(newVal ? 'نمایش داده شد' : 'از نمایش خارج شد');
   347→    } catch {
   348→      toast.error('خطا');
   349→      loadProducts();
   350→    }
   351→  };
   352→
   353→  const availabilityLabel = (a: string) => {
   354→    if (a === 'in_stock') return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100">موجود</Badge>;
   355→    if (a === 'out_of_stock') return <Badge className="bg-red-100 text-red-700 hover:bg-red-100">ناموجود</Badge>;
   356→    return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100">تماس بگیرید</Badge>;
   357→  };
   358→
   359→  const totalPages = Math.ceil(total / limit);
   360→
   361→  return (
   362→    <div className="space-y-4" dir="rtl">
   363→      {/* Toolbar */}
   364→      <div className="flex flex-col sm:flex-row gap-3 items-stretch sm:items-center">
   365→        <div className="relative flex-1 max-w-md">
   366→          <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   367→          <Input
   368→            placeholder="جستجوی محصول..."
   369→            value={q}
   370→            onChange={e => handleSearch(e.target.value)}
   371→            className="pr-10"
   372→          />
   373→        </div>
   374→        <Button onClick={openCreate} className="bg-[#16233F] hover:bg-[#1e2d4d]">
   375→          <Plus className="w-4 h-4 ml-1" /> افزودن محصول
   376→        </Button>
   377→      </div>
   378→
   379→      {/* Table */}
   380→      <Card>
   381→        <CardContent className="p-0">
   382→          {loading ? (
   383→            <div className="p-6 space-y-3">
   384→              {[1,2,3,4,5].map(i => <Skeleton key={i} className="h-12 w-full" />)}
   385→            </div>
   386→          ) : (
   387→            <div className="overflow-x-auto">
   388→              <table className="w-full text-sm">
   389→                <thead>
   390→                  <tr className="border-b bg-gray-50 text-muted-foreground">
   391→                    <th className="text-right py-3 px-3 font-medium">شناسه</th>
   392→                    <th className="text-right py-3 px-3 font-medium">نام محصول</th>
   393→                    <th className="text-right py-3 px-3 font-medium hidden md:table-cell">برند</th>
   394→                    <th className="text-right py-3 px-3 font-medium hidden lg:table-cell">دسته‌بندی</th>
   395→                    <th className="text-right py-3 px-3 font-medium">قیمت</th>
   396→                    <th className="text-right py-3 px-3 font-medium">وضعیت</th>
   397→                    <th className="text-center py-3 px-3 font-medium">تصاویر</th>
   398→                    <th className="text-center py-3 px-3 font-medium">ویژه</th>
   399→                    <th className="text-center py-3 px-3 font-medium">نمایش</th>
   400→                    <th className="text-center py-3 px-3 font-medium">عملیات</th>
   401→                  </tr>
   402→                </thead>
   403→                <tbody>
   404→                  {products.length === 0 && (
   405→                    <tr><td colSpan={10} className="text-center py-12 text-muted-foreground">محصولی یافت نشد</td></tr>
   406→                  )}
   407→                  {products.map(p => {
   408→                    const isVisible = p.visible !== false;
   409→                    return (
   410→                    <tr key={p.id} className={`border-b last:border-0 hover:bg-gray-50/50 ${!isVisible ? 'bg-gray-50/60' : ''}`}>
   411→                      <td className="py-2.5 px-3 text-muted-foreground">#{p.id}</td>
   412→                      <td className={`py-2.5 px-3 font-medium max-w-[200px] truncate ${!isVisible ? 'text-gray-400' : ''}`}>{p.name}{!isVisible && <span className="block text-[10px] text-gray-400">پنهان</span>}</td>
   413→                      <td className="py-2.5 px-3 hidden md:table-cell text-gray-600">{p.brand.name}</td>
   414→                      <td className="py-2.5 px-3 hidden lg:table-cell text-gray-600">{p.category.name}</td>
   415→                      <td className="py-2.5 px-3">
   416→                        {p.price ? formatNumber(p.price) + ' ت' : <span className="text-muted-foreground">تماس</span>}
   417→                      </td>
   418→                      <td className="py-2.5 px-3">{availabilityLabel(p.availability)}</td>
   419→                      <td className="py-2.5 px-3 text-center">
   420→                        <button onClick={() => openImages(p)} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-[#2FC4C9] transition-colors">
   421→                          <ImageIcon className="w-3.5 h-3.5" /> {p._count.images}
   422→                        </button>
   423→                      </td>
   424→                      <td className="py-2.5 px-3 text-center">
   425→                        <Switch checked={p.featured} onCheckedChange={() => toggleFeatured(p)} className="mx-auto" />
   426→                      </td>
   427→                      <td className="py-2.5 px-3 text-center">
   428→                        <Switch checked={isVisible} onCheckedChange={() => toggleVisible(p)} className="mx-auto" />
   429→                      </td>
   430→                      <td className="py-2.5 px-3 text-center">
   431→                        <div className="flex items-center justify-center gap-1">
   432→                          <button onClick={() => openEdit(p)} className="p-1.5 rounded-md hover:bg-blue-50 text-blue-600 transition-colors">
   433→                            <Pencil className="w-4 h-4" />
   434→                          </button>
   435→                          <button onClick={() => setDeleteTarget(p)} className="p-1.5 rounded-md hover:bg-red-50 text-red-500 transition-colors">
   436→                            <Trash2 className="w-4 h-4" />
   437→                          </button>
   438→                        </div>
   439→                      </td>
   440→                    </tr>
   441→                    );
   442→                  })}
   443→                </tbody>
   444→              </table>
   445→            </div>
   446→          )}
   447→        </CardContent>
   448→      </Card>
   449→
   450→      {/* Pagination */}
   451→      {totalPages > 1 && (
   452→        <div className="flex items-center justify-center gap-2">
   453→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(p => p - 1)}>
   454→            <ChevronRight className="w-4 h-4" />
   455→          </Button>
   456→          <span className="text-sm text-muted-foreground">
   457→            صفحه {page} از {totalPages}
   458→          </span>
   459→          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(p => p + 1)}>
   460→            <ChevronLeft className="w-4 h-4" />
   461→          </Button>
   462→        </div>
   463→      )}
   464→
   465→      {/* Create/Edit Dialog */}
   466→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   467→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto" dir="rtl">
   468→          <DialogHeader>
   469→            <DialogTitle>{editing ? 'ویرایش محصول' : 'افزودن محصول جدید'}</DialogTitle>
