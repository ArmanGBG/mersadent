   202→const formatPrice = (price: number | null) => {
   203→  if (price === null) return 'استعلام قیمت';

   207→const useDebounce = (value: string, delay: number) => {
   208→  const [debounced, setDebounced] = useState(value);
   209→  useEffect(() => {
   210→    const t = setTimeout(() => setDebounced(value), delay);
   211→    return () => clearTimeout(t);
   212→  }, [value, delay]);
   213→  return debounced;
   214→};

   261→function ProductCard({ product, onClick, user, onAuthRequired, onAddToCart, onToggleCompare, isInCompare }: { product: Product; onClick: () => void; user: UserData | null; onAuthRequired: () => void; onAddToCart: (productId: number) => void; onToggleCompare: () => void; isInCompare: boolean }) {
   262→  const av = availabilityLabel(product.availability);
   263→  const hasDiscount = product.discountPercent > 0 && product.price !== null && product.priceType !== 'call';
   264→  const discountedPrice = hasDiscount ? Math.round(product.price! * (1 - product.discountPercent / 100)) : null;
   265→  const canAdd = product.availability === 'in_stock' && product.price !== null && product.priceType !== 'call';
   266→  const [adding, setAdding] = useState(false);
   267→
   268→  const handleAddToCart = async (e: React.MouseEvent) => {
   269→    e.stopPropagation();
   270→    if (!user) { onAuthRequired(); return; }
   271→    if (adding) return;
   272→    setAdding(true);
   273→    await onAddToCart(product.id);
   274→    setAdding(false);
   275→  };
   276→
   277→  return (
   278→    <motion.div whileHover={{ y: -4 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   279→      <Card
   280→        className="cursor-pointer group h-full border border-gray-100 hover:border-mersadent-accent/40 hover:shadow-lg transition-all duration-300 overflow-hidden"
   281→        onClick={onClick}
   282→      >
   283→        <CardContent className="p-3 flex flex-col gap-2">
   284→          {/* Image or placeholder */}
   285→          <div className="relative w-full aspect-square bg-gradient-to-br from-gray-50 to-gray-100 rounded-lg flex items-center justify-center group-hover:from-mersadent-accent/5 group-hover:to-mersadent-accent/10 transition-colors overflow-hidden">
   286→            {product.images && product.images.length > 0 ? (
   287→              <img
   288→                src={product.images[0].url}
   289→                alt={product.images[0].alt || product.name}
   290→                className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
   291→                loading="lazy"
   292→              />
   293→            ) : (
   294→              <Package className="w-10 h-10 text-gray-300 group-hover:text-mersadent-accent/50 transition-colors" />
   295→            )}
   296→            {/* Top badges: featured + discount */}
   297→            <div className="absolute top-1.5 left-1.5 flex flex-col gap-1">
   298→              {hasDiscount && (
   299→                <Badge className="bg-red-500 text-white text-[9px] px-1.5 py-0">
   300→                  تخفیف {product.discountPercent}%
   301→                </Badge>
   302→              )}
   303→              {product.featured && (
   304→                <Badge className="bg-amber-500 text-white text-[9px] px-1 py-0">
   305→                  <Star className="w-2.5 h-2.5 ml-0.5" />
   306→                  ویژه
   307→                </Badge>
   308→              )}
   309→            </div>
   310→            {/* Compare toggle button */}
   311→            <button
   312→              onClick={(e) => { e.stopPropagation(); onToggleCompare(); }}
   313→              className={`absolute bottom-1.5 left-1.5 w-7 h-7 rounded-full flex items-center justify-center transition-all duration-200 z-10 ${isInCompare ? 'bg-teal-500 text-white shadow-md opacity-100' : 'bg-white/90 text-gray-400 hover:text-teal-500 hover:bg-white shadow-sm opacity-80 group-hover:opacity-100'}`}
   314→              title={isInCompare ? 'حذف از لیست مقایسه' : 'افزودن به لیست مقایسه'}
   315→              aria-label={isInCompare ? 'حذف از مقایسه' : 'افزودن به مقایسه'}
   316→              aria-pressed={isInCompare}
   317→            >
   318→              <GitCompareArrows className="w-3.5 h-3.5" />
   319→            </button>
   320→            {/* Add to cart button */}
   321→            {canAdd && (
   322→              <button
   323→                onClick={handleAddToCart}
   324→                disabled={adding}
   325→                className="absolute bottom-1.5 left-9 w-7 h-7 bg-white/90 hover:bg-mersadent-accent text-mersadent-dark hover:text-white rounded-full flex items-center justify-center shadow-md opacity-0 group-hover:opacity-100 transition-all duration-200 disabled:opacity-60"
   326→                title="افزودن به سبد خرید"
   327→              >
   328→                {adding ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <ShoppingCart className="w-3.5 h-3.5" />}
   329→              </button>
   330→            )}
   331→            {product.availability === 'out_of_stock' && (
   332→              <div className="absolute inset-0 bg-white/70 rounded-lg flex items-center justify-center">
   333→                <Badge variant="secondary" className="text-red-500 text-xs">ناموجود</Badge>
   334→              </div>
   335→            )}
   336→            {/* Low stock indicator */}
   337→            {product.lowStock > 0 && product.availability === 'in_stock' && (
   338→              <Badge className="absolute bottom-1.5 right-1.5 bg-red-500 text-white text-[9px] px-1.5 py-0 font-bold">
   339→                فقط {new Intl.NumberFormat('fa-IR').format(product.lowStock)} عدد مانده
   340→              </Badge>
   341→            )}
   342→            {/* Product tags */}
   343→            {product.tags && product.tags.length > 0 && (
   344→              <div className="absolute top-1.5 right-1.5 flex flex-col gap-1">
   345→                {product.tags.filter(t => t.tag.type === 'badge' || t.tag.type === 'consumable').map((t) => (
   346→                  <Badge
   347→                    key={t.tag.id}
   348→                    className={`text-[8px] px-1.5 py-0 font-medium ${
   349→                      t.tag.type === 'consumable'
   350→                        ? 'bg-orange-100 text-orange-700 border-orange-200'
   351→                        : 'bg-mersadent-accent/10 text-mersadent-accent border-mersadent-accent/20'
   352→                    }`}
   353→                  >
   354→                    {t.tag.name}
   355→                  </Badge>
   356→                ))}
   357→              </div>
   358→            )}
   359→          </div>
   360→
   361→          {/* Info */}
   362→          <div className="flex-1 flex flex-col gap-1 min-h-0">
   363→            <span className="text-[10px] text-gray-400 font-medium">{product.brand.name}</span>
   364→            <h3 className="text-xs font-semibold text-mersadent-dark leading-5 line-clamp-2">
   365→              {product.name}
   366→            </h3>
   367→          </div>
   368→
   369→          {/* Price & availability */}
   370→          <div className="flex flex-col gap-1 pt-1.5 border-t border-gray-100">
   371→            {product.priceType === 'call' || product.price === null ? (
   372→              <Badge className="w-fit bg-mersadent-accent/10 text-mersadent-accent border-mersadent-accent/20 text-xs font-medium">
   373→                تماس بگیرید
   374→              </Badge>
   375→            ) : hasDiscount && discountedPrice !== null ? (
   376→              <div className="flex flex-col gap-0.5">
   377→                <span className="text-[11px] text-gray-400 line-through">
   378→                  {formatPrice(product.price)}
   379→                </span>
   380→                <span className="text-sm font-bold text-red-600">
   381→                  {formatPrice(discountedPrice)}
   382→                </span>
   383→              </div>
   384→            ) : (
   385→              <span className="text-sm font-bold text-mersadent-dark">
   386→                {formatPrice(product.price)}
   387→              </span>
   388→            )}
   389→            <span className={`text-[11px] px-2 py-0.5 rounded-full w-fit ${av.color}`}>
   390→              {av.text}
   391→            </span>
   392→          </div>
   393→        </CardContent>
   394→      </Card>
   395→    </motion.div>
   396→  );
   397→}

   437→        setSearchResults([]);
   438→        setSearchOpen(false);

   440→      return;

   445→        setSearchResults(d.results || []);
   446→        setSearchOpen(true);

   454→        setSearchOpen(false);

   470→      setSearchOpen(false);
   471→      onSearch(searchQuery.trim());
   472→    }

   476→    { label: 'مقالات آموزشی', action: () => { setMobileOpen(false); onNavigate({ type: 'blog' }); } },
   477→    { label: 'برندها', action: () => { setMobileOpen(false); onNavigate({ type: 'brand', brandId: 0, brandName: 'all' }); } },
   478→    { label: 'درباره ما', action: () => { setMobileOpen(false); onNavigate({ type: 'about' }); } },
   479→  ];
   480→
   481→  // Mega menu hover handlers with delay
   482→  const handleCatEnter = (cat: Category | null) => {
   483→    if (catTimeoutRef.current) clearTimeout(catTimeoutRef.current);
   484→    setActiveCat(cat);
   485→  };
   486→  const handleCatLeave = () => {
   487→    catTimeoutRef.current = setTimeout(() => {
   488→      setActiveCat(null);
   489→      setCatMenuOpen(false);
   490→    }, 200);
   491→  };
   492→
   493→  return (
   494→    <header className="sticky top-0 z-50 bg-white border-b border-gray-200/80 shadow-sm">
   495→      <div className="max-w-7xl mx-auto px-4 sm:px-6">
   496→        <div className="flex items-center justify-between h-14 gap-4">
   497→          {/* Logo + Brand Name */}
   498→          <button
   499→            onClick={() => onNavigate({ type: 'home' })}
   500→            className="flex-shrink-0 flex items-center gap-2.5 cursor-pointer group"
   501→          >
   502→            <img src="/logo.png" alt="MersaDent" className="h-8 sm:h-9 w-auto object-contain" />
   503→            <div className="flex flex-col leading-none">
   504→              <span className="text-xl sm:text-2xl font-extrabold tracking-tight text-mersadent-dark group-hover:text-mersadent-accent transition-colors" style={{ fontFamily: "'Geist Mono', monospace" }}>MersaDent</span>
   505→              <span className="text-[11px] sm:text-xs font-bold text-mersadent-accent/70 mt-0.5">مرصادنت</span>
   506→            </div>
   507→          </button>
   508→
   509→          {/* Desktop Nav with Mega Menu */}
   510→          <nav className="hidden md:flex items-center gap-1">
   511→            {/* Categories Mega Menu */}
   512→            <div
   513→              ref={megaMenuRef}
   514→              className="relative"
   515→              onMouseEnter={() => { setCatMenuOpen(true); if (catTimeoutRef.current) clearTimeout(catTimeoutRef.current); }}
   516→              onMouseLeave={handleCatLeave}
   517→            >
   518→              <button
   519→                className={`flex items-center gap-1.5 text-sm font-medium px-3 py-2 rounded-lg transition-colors ${catMenuOpen ? 'text-mersadent-accent bg-mersadent-accent/5' : 'text-gray-600 hover:text-mersadent-accent'}`}
   520→              >
   521→                دسته‌بندی‌ها
   522→                <ChevronDown className={`w-3.5 h-3.5 transition-transform ${catMenuOpen ? 'rotate-180' : ''}`} />

   858→

   860→    if (isTransitioning) return;
   861→    setIsTransitioning(true);
   862→    setCurrent(index);
   863→    currentRef.current = index;
   864→    setTimeout(() => setIsTransitioning(false), 700);
   865→  }, [isTransitioning]);
   866→
   867→  const next = useCallback(() => goTo((current + 1) % heroSlides.length), [current, goTo]);

   869→

   871→  useEffect(() => {
   872→    intervalRef.current = setInterval(() => {
   873→      goTo((currentRef.current + 1) % heroSlides.length);
   874→    }, 6000);
   875→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   876→  }, [goTo, heroSlides.length]);
   877→

   879→  const pause = () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   880→  const resume = () => {
   881→    intervalRef.current = setInterval(() => {
   882→      goTo((currentRef.current + 1) % heroSlides.length);
   883→    }, 6000);
   884→  };
   885→
   886→  // Touch handlers for swipe

   888→  const onTouchEnd = (e: React.TouchEvent) => {
   889→    const diff = touchStartX.current - e.changedTouches[0].clientX;
   890→    if (Math.abs(diff) > 50) { if (diff > 0) next(); else prev(); }
   891→  };
   892→

   897→      className="relative w-full h-[50vw] sm:h-[42vw] md:h-[36vw] lg:h-[520px] xl:h-[560px] min-h-[280px] max-h-[600px] overflow-hidden select-none"

   899→      onMouseLeave={resume}

   901→      onTouchEnd={onTouchEnd}

   906→          key={i}
   907→          className="absolute inset-0 transition-opacity duration-700 ease-in-out"

   912→            alt={s.title}
   913→            className="w-full h-full object-cover"
   914→            style={{ objectPosition: s.objectPosition }}
   915→            loading={i === 0 ? 'eager' : 'lazy'}
   916→          />
   917→        </div>
   918→      ))}
   919→
   920→      {/* Dark gradient overlay */}
   921→      <div className="absolute inset-0 bg-gradient-to-l from-black/70 via-black/50 to-transparent z-10" />
   922→
   923→      {/* Content */}
   924→      <div className="absolute inset-0 z-20 flex items-center">
   925→        <div className="max-w-7xl mx-auto w-full px-5 sm:px-8 lg:px-12">

   928→              key={current}
   929→              initial={{ opacity: 0, y: 20 }}

   931→              exit={{ opacity: 0, y: -15 }}

   944→                  className="bg-mersadent-accent hover:bg-mersadent-accent/90 text-white rounded-xl px-5 sm:px-6 h-10 sm:h-11 font-semibold text-sm sm:text-base"
