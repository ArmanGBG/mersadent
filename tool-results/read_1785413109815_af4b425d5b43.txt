     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Input } from '@/components/ui/input';
     6→import { Label } from '@/components/ui/label';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Card, CardContent } from '@/components/ui/card';
     9→import { Skeleton } from '@/components/ui/skeleton';
    10→import { Switch } from '@/components/ui/switch';
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    12→import { Textarea } from '@/components/ui/textarea';
    13→import { Separator } from '@/components/ui/separator';
    14→import {
    15→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription,
    16→} from '@/components/ui/dialog';
    17→import {
    18→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    19→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    20→} from '@/components/ui/alert-dialog';
    21→import {
    22→  Search, Plus, Pencil, Trash2, Upload, X, ChevronLeft, ChevronRight, ImageIcon, Loader2, Wand2, FileText, CheckCircle2, AlertTriangle
    23→} from 'lucide-react';
    24→import { toast } from 'sonner';
    25→import { adminFetch, formatNumber, toSlug } from '@/lib/admin-utils';
    26→import { parseSuperPrompt } from '@/lib/parse-super-prompt';
    27→
    28→interface AdminProduct {
    29→  id: number; name: string; slug: string; model: string | null;
    30→  price: number | null; priceType: string; availability: string;
    31→  specs: string | null; featured: boolean; visible: boolean; order: number; discountPercent: number;
    32→  description: string | null;
    33→  strengths: string | null;
    34→  weaknesses: string | null;
    35→  metaTitle: string | null;
    36→  metaDescription: string | null;
    37→  keywords: string | null;
    38→  sku: string | null;
    39→  country: string | null;
    40→  warranty: string | null;
    41→  weight: string | null;
    42→  lowStock: number;
    43→  category: { id: number; name: string; slug: string };
    44→  brand: { id: number; name: string; slug: string };
    45→  _count: { images: number };
    46→  tags: { tagId: number; tag: { id: number; name: string; slug: string; type: string } }[];
    47→}
    48→
    49→interface SimpleCategory { id: number; name: string; slug: string; parentId: number | null; }
    50→interface SimpleBrand { id: number; name: string; slug: string; }
    51→
    52→interface SimpleTag { id: number; name: string; slug: string; type: string; }
    53→
    54→interface ProductForm {
    55→  name: string; slug: string; model: string; price: string; priceType: string;
    56→  availability: string; specs: string; categoryId: string; brandId: string;
    57→  featured: boolean; visible: boolean; discountPercent: string;
    58→  description: string; strengths: string[]; weaknesses: string[];
    59→  metaTitle: string; metaDescription: string; keywords: string;
    60→  sku: string; country: string; warranty: string; weight: string; lowStock: string;
    61→  tagIds: number[];
    62→}
    63→
    64→const emptyProduct: ProductForm = {
    65→  name: '', slug: '', model: '', price: '', priceType: 'fixed', availability: 'in_stock',
    66→  specs: '', categoryId: '', brandId: '', featured: false, visible: true, discountPercent: '0',
    67→  description: '', strengths: [], weaknesses: [], metaTitle: '', metaDescription: '', keywords: '',
    68→  sku: '', country: '', warranty: '', weight: '', lowStock: '0', tagIds: [],
    69→};
    70→
    71→export function AdminProducts({ token }: { token: string }) {
    72→  const [products, setProducts] = useState<AdminProduct[]>([]);
    73→  const [total, setTotal] = useState(0);
    74→  const [page, setPage] = useState(1);
    75→  const [q, setQ] = useState('');
    76→  const [loading, setLoading] = useState(true);
    77→  const [searchTimeout, setSearchTimeout] = useState<NodeJS.Timeout | null>(null);
    78→
    79→  const [categories, setCategories] = useState<SimpleCategory[]>([]);
    80→  const [brands, setBrands] = useState<SimpleBrand[]>([]);
    81→  const [allTags, setAllTags] = useState<SimpleTag[]>([]);
    82→
    83→  // Dialog state
    84→  const [dialogOpen, setDialogOpen] = useState(false);
    85→  const [editing, setEditing] = useState<AdminProduct | null>(null);
    86→  const [form, setForm] = useState<ProductForm>({ ...emptyProduct });
    87→  const [saving, setSaving] = useState(false);
    88→
    89→  // Super-prompt state
    90→  const [superPromptText, setSuperPromptText] = useState('');
    91→  const [parsing, setParsing] = useState(false);
    92→
    93→  // Delete state
    94→  const [deleteTarget, setDeleteTarget] = useState<AdminProduct | null>(null);
    95→  const [deleting, setDeleting] = useState(false);
    96→
    97→  // Image dialog
    98→  const [imageProduct, setImageProduct] = useState<AdminProduct | null>(null);
    99→  const [images, setImages] = useState<{ id: number; url: string; alt: string | null; order: number }[]>([]);
   100→  const [uploading, setUploading] = useState(false);
   101→
   102→  const limit = 20;
   103→
   104→  const loadProducts = useCallback(async () => {
   105→    setLoading(true);
   106→    try {
   107→      const res = await adminFetch(token, `/api/admin/products?page=${page}&limit=${limit}&q=${encodeURIComponent(q)}`);
   108→      const data = await res.json();
   109→      if (data.error) { toast.error(data.error); return; }
   110→      setProducts(data.products);
   111→      setTotal(data.total);
   112→    } catch { toast.error('خطا در دریافت محصولات'); }
   113→    finally { setLoading(false); }
   114→  }, [token, page, q]);
   115→
   116→  const loadLookups = useCallback(async () => {
   117→    try {
   118→      const [catRes, brandRes, tagsRes] = await Promise.all([
   119→        adminFetch(token, '/api/admin/categories?flat=true'),
   120→        adminFetch(token, '/api/admin/brands'),
   121→        fetch('/api/tags'),
   122→      ]);
   123→      const catData = await catRes.json();
   124→      const brandData = await brandRes.json();
   125→      const tagsData = await tagsRes.json();
   126→      if (catData.categories) setCategories(catData.categories);
   127→      if (brandData.brands) setBrands(brandData.brands);
   128→      if (Array.isArray(tagsData)) setAllTags(tagsData);
   129→    } catch { /* ignore */ }
   130→  }, [token]);
   131→
   132→  useEffect(() => { loadLookups(); }, [loadLookups]);
   133→  useEffect(() => { loadProducts(); }, [loadProducts]);
   134→
   135→  const handleSearch = (value: string) => {
   136→    setQ(value);
   137→    setPage(1);
   138→    if (searchTimeout) clearTimeout(searchTimeout);
   139→    setSearchTimeout(setTimeout(() => {}, 100));
   140→  };
   141→
   142→  const openCreate = () => {
   143→    setEditing(null);
   144→    setForm({ ...emptyProduct });
   145→    setSuperPromptText('');
   146→    setDialogOpen(true);
   147→  };
   148→
   149→  const openEdit = (p: AdminProduct) => {
   150→    setEditing(p);
   151→    setForm({
   152→      name: p.name,
   153→      slug: p.slug,
   154→      model: p.model || '',
   155→      price: p.price !== null ? String(p.price) : '',
   156→      priceType: p.priceType,
   157→      availability: p.availability,
   158→      specs: p.specs || '',
   159→      categoryId: String(p.category.id),
   160→      brandId: String(p.brand.id),
   161→      featured: p.featured,
   162→      visible: p.visible !== false,
   163→      discountPercent: String(p.discountPercent ?? 0),
   164→      description: p.description || '',
   165→      strengths: p.strengths ? (() => { try { return JSON.parse(p.strengths); } catch { return []; } })() : [],
   166→      weaknesses: p.weaknesses ? (() => { try { return JSON.parse(p.weaknesses); } catch { return []; } })() : [],
   167→      metaTitle: p.metaTitle || '',
   168→      metaDescription: p.metaDescription || '',
   169→      keywords: p.keywords || '',
   170→      sku: p.sku || '',
   171→      country: p.country || '',
   172→      warranty: p.warranty || '',
   173→      weight: p.weight || '',
   174→      lowStock: String(p.lowStock ?? 0),
   175→      tagIds: (p.tags || []).map((pt) => pt.tag.id),
   176→    });
   177→    setSuperPromptText('');
   178→    setDialogOpen(true);
   179→  };
   180→
   181→  const handleSave = async () => {
   182→    if (!form.name || !form.categoryId || !form.brandId) {
   183→      toast.error('نام، دسته‌بندی و برند الزامی هستند');
   184→      return;
   185→    }
   186→    setSaving(true);
   187→    try {
   188→      const body: Record<string, unknown> = {
   189→        name: form.name,
   190→        slug: form.slug || toSlug(form.name),
   191→        model: form.model || null,
   192→        price: form.price === '' ? null : Number(form.price),
   193→        discountPercent: Number(form.discountPercent) || 0,
   194→        priceType: form.priceType,
   195→        availability: form.availability,
   196→        specs: form.specs || null,
   197→        categoryId: Number(form.categoryId),
   198→        brandId: Number(form.brandId),
   199→        featured: form.featured,
   200→        visible: form.visible,
   201→        description: form.description || null,
   202→        strengths: form.strengths.length > 0 ? JSON.stringify(form.strengths) : null,
   203→        weaknesses: form.weaknesses.length > 0 ? JSON.stringify(form.weaknesses) : null,
   204→        metaTitle: form.metaTitle || null,
   205→        metaDescription: form.metaDescription || null,
   206→        keywords: form.keywords || null,
   207→        sku: form.sku || null,
   208→        country: form.country || null,
   209→        warranty: form.warranty || null,
   210→        weight: form.weight || null,
   211→        lowStock: form.lowStock || '0',
   212→        tagIds: form.tagIds,
   213→      };
   214→
   215→      if (editing) {
   216→        body.id = editing.id;
   217→        const res = await adminFetch(token, '/api/admin/products', { method: 'PUT', body: JSON.stringify(body) });
   218→        const data = await res.json();
   219→        if (data.error) { toast.error(data.error); return; }
   220→        toast.success('محصول بروزرسانی شد');
   221→      } else {
   222→        const res = await adminFetch(token, '/api/admin/products', { method: 'POST', body: JSON.stringify(body) });
   223→        const data = await res.json();
   224→        if (data.error) { toast.error(data.error); return; }
   225→        toast.success('محصول ایجاد شد');
   226→      }
   227→      setDialogOpen(false);
   228→      loadProducts();
   229→    } catch { toast.error('خطا در ذخیره محصول'); }
   230→    finally { setSaving(false); }
   231→  };
   232→
   233→  const handleParseSuperPrompt = () => {
   234→    if (!superPromptText.trim()) {
   235→      toast.error('ابتدا متن خروجی را وارد کنید');
   236→      return;
   237→    }
   238→    setParsing(true);
   239→    // Use setTimeout to allow UI to update before synchronous parse
   240→    setTimeout(() => {
   241→      try {
   242→        const parsed = parseSuperPrompt(superPromptText);
   243→        setForm(f => ({
   244→          ...f,
   245→          description: parsed.description || f.description,
   246→          strengths: parsed.strengths.length > 0 ? parsed.strengths : f.strengths,
   247→          weaknesses: parsed.weaknesses.length > 0 ? parsed.weaknesses : f.weaknesses,
   248→          metaTitle: parsed.metaTitle || f.metaTitle,
   249→          metaDescription: parsed.metaDescription || f.metaDescription,
   250→          keywords: parsed.keywords || f.keywords,
   251→        }));
   252→        const foundCount = [
   253→          parsed.description ? 'توضیحات' : '',
   254→          parsed.strengths.length ? `${parsed.strengths.length} نقطه قوت` : '',
   255→          parsed.weaknesses.length ? `${parsed.weaknesses.length} نقطه ضعف` : '',
   256→        ].filter(Boolean);
   257→        if (foundCount.length > 0) {
   258→          toast.success(`تحلیل موفق: ${foundCount.join('، ')}`);
   259→        } else {
   260→          toast.warning('متن پردازش شد اما بخش مشخصی یافت نشد');
   261→        }
   262→      } catch (e) {
   263→        toast.error('خطا در تحلیل متن');
   264→        console.error('Parse error:', e);
   265→      } finally {
   266→        setParsing(false);
   267→      }
   268→    }, 50);
   269→  };
   270→
   271→  const removeStrength = (idx: number) => {
   272→    setForm(f => ({ ...f, strengths: f.strengths.filter((_, i) => i !== idx) }));
   273→  };
   274→
   275→  const removeWeakness = (idx: number) => {
   276→    setForm(f => ({ ...f, weaknesses: f.weaknesses.filter((_, i) => i !== idx) }));
   277→  };
   278→
   279→  const handleDelete = async () => {
   280→    if (!deleteTarget) return;
   281→    setDeleting(true);
   282→    try {
   283→      const res = await adminFetch(token, `/api/admin/products?id=${deleteTarget.id}`, { method: 'DELETE' });
   284→      const data = await res.json();
   285→      if (data.error) { toast.error(data.error); return; }
   286→      toast.success('محصول حذف شد');
   287→      setDeleteTarget(null);
   288→      loadProducts();
   289→    } catch { toast.error('خطا در حذف محصول'); }
   290→    finally { setDeleting(false); }
   291→  };
   292→
   293→  const openImages = async (p: AdminProduct) => {
   294→    setImageProduct(p);
   295→    try {
   296→      const res = await adminFetch(token, `/api/admin/products/${p.id}/images`);
   297→      const data = await res.json();
   298→      setImages(data.images || []);
   299→    } catch { setImages([]); }
   300→  };
   301→
   302→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   303→    const file = e.target.files?.[0];
   304→    if (!file || !imageProduct) return;
   305→    setUploading(true);
   306→    try {
   307→      const fd = new FormData();
   308→      fd.append('file', file);
   309→      fd.append('productId', String(imageProduct.id));
   310→      const res = await fetch('/api/admin/upload', { method: 'POST', body: fd });
   311→      const data = await res.json();
   312→      if (data.error) { toast.error(data.error); return; }
   313→      toast.success('تصویر آپلود شد');
   314→      openImages(imageProduct);
   315→    } catch { toast.error('خطا در آپلود'); }
   316→    finally { setUploading(false); e.target.value = ''; }
   317→  };
   318→
   319→  const handleDeleteImage = async (imageId: number) => {
   320→    if (!imageProduct) return;
   321→    try {
   322→      const res = await adminFetch(token, `/api/admin/products/${imageProduct.id}/images?imageId=${imageId}`, { method: 'DELETE' });
   323→      const data = await res.json();
   324→      if (data.error) { toast.error(data.error); return; }
   325→      toast.success('تصویر حذف شد');
   326→      setImages(prev => prev.filter(i => i.id !== imageId));
   327→    } catch { toast.error('خطا در حذف تصویر'); }
   328→  };
   329→
   330→  const toggleFeatured = async (p: AdminProduct) => {
   331→    try {
   332→      const res = await adminFetch(token, '/api/admin/products', {
   333→        method: 'PUT',
   334→        body: JSON.stringify({ id: p.id, name: p.name, featured: !p.featured }),
   335→      });
   336→      const data = await res.json();
   337→      if (data.error) { toast.error(data.error); return; }
   338→      toast.success(p.featured ? 'از ویژه‌ها حذف شد' : 'به ویژه‌ها اضافه شد');
   339→      loadProducts();
   340→    } catch { toast.error('خطا'); }
   341→  };
   342→
   343→  const toggleVisible = async (p: AdminProduct) => {
   344→    const newVal = !(p.visible !== false);
   345→    // Optimistic
   346→    setProducts(prev => prev.map(x => x.id === p.id ? { ...x, visible: newVal } : x));
   347→    try {
   348→      const res = await adminFetch(token, '/api/admin/products', {
   349→        method: 'PUT',
   350→        body: JSON.stringify({ id: p.id, name: p.name, visible: newVal }),
   351→      });
   352→      const data = await res.json();
   353→      if (data.error) {
   354→        toast.error(data.error);
   355→        loadProducts();
   356→        return;
   357→      }
   358→      toast.success(newVal ? 'نمایش داده شد' : 'از نمایش خارج شد');
   359→    } catch {
   360→      toast.error('خطا');
   361→      loadProducts();
   362→    }
   363→  };
   364→
   365→  const availabilityLabel = (a: string) => {
   366→    if (a === 'in_stock') return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100">موجود</Badge>;
   367→    if (a === 'out_of_stock') return <Badge className="bg-red-100 text-red-700 hover:bg-red-100">ناموجود</Badge>;
   368→    return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100">تماس بگیرید</Badge>;
   369→  };
   370→
   371→  const totalPages = Math.ceil(total / limit);
   372→
   373→  return (
   374→    <div className="space-y-4" dir="rtl">
   375→      {/* Toolbar */}
   376→      <div className="flex flex-col sm:flex-row gap-3 items-stretch sm:items-center">
   377→        <div className="relative flex-1 max-w-md">
   378→          <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   379→          <Input
   380→            placeholder="جستجوی محصول..."
   381→            value={q}
   382→            onChange={e => handleSearch(e.target.value)}
   383→            className="pr-10"
   384→          />
   385→        </div>
   386→        <Button onClick={openCreate} className="bg-[#16233F] hover:bg-[#1e2d4d]">
   387→          <Plus className="w-4 h-4 ml-1" /> افزودن محصول
   388→        </Button>
   389→      </div>
   390→
   391→      {/* Table */}
   392→      <Card>
   393→        <CardContent className="p-0">
   394→          {loading ? (
   395→            <div className="p-6 space-y-3">
   396→              {[1,2,3,4,5].map(i => <Skeleton key={i} className="h-12 w-full" />)}
   397→            </div>
   398→          ) : (
   399→            <div className="overflow-x-auto">
   400→              <table className="w-full text-sm">
   401→                <thead>
   402→                  <tr className="border-b bg-gray-50 text-muted-foreground">
   403→                    <th className="text-right py-3 px-3 font-medium">شناسه</th>
   404→                    <th className="text-right py-3 px-3 font-medium">نام محصول</th>
   405→                    <th className="text-right py-3 px-3 font-medium hidden md:table-cell">برند</th>
   406→                    <th className="text-right py-3 px-3 font-medium hidden lg:table-cell">دسته‌بندی</th>
   407→                    <th className="text-right py-3 px-3 font-medium">قیمت</th>
   408→                    <th className="text-right py-3 px-3 font-medium">وضعیت</th>
   409→                    <th className="text-center py-3 px-3 font-medium">تصاویر</th>
   410→                    <th className="text-center py-3 px-3 font-medium">ویژه</th>
   411→                    <th className="text-center py-3 px-3 font-medium">نمایش</th>
   412→                    <th className="text-center py-3 px-3 font-medium">عملیات</th>
   413→                  </tr>
   414→                </thead>
   415→                <tbody>
   416→                  {products.length === 0 && (
   417→                    <tr><td colSpan={10} className="text-center py-12 text-muted-foreground">محصولی یافت نشد</td></tr>
   418→                  )}
   419→                  {products.map(p => {
   420→                    const isVisible = p.visible !== false;
   421→                    return (
   422→                    <tr key={p.id} className={`border-b last:border-0 hover:bg-gray-50/50 ${!isVisible ? 'bg-gray-50/60' : ''}`}>
   423→                      <td className="py-2.5 px-3 text-muted-foreground">#{p.id}</td>
   424→                      <td className={`py-2.5 px-3 font-medium max-w-[200px] truncate ${!isVisible ? 'text-gray-400' : ''}`}>{p.name}{!isVisible && <span className="block text-[10px] text-gray-400">پنهان</span>}</td>
   425→                      <td className="py-2.5 px-3 hidden md:table-cell text-gray-600">{p.brand.name}</td>
   426→                      <td className="py-2.5 px-3 hidden lg:table-cell text-gray-600">{p.category.name}</td>
   427→                      <td className="py-2.5 px-3">
   428→                        {p.price ? formatNumber(p.price) + ' ت' : <span className="text-muted-foreground">تماس</span>}
   429→                      </td>
   430→                      <td className="py-2.5 px-3">{availabilityLabel(p.availability)}</td>
   431→                      <td className="py-2.5 px-3 text-center">
   432→                        <button onClick={() => openImages(p)} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-[#2FC4C9] transition-colors">
   433→                          <ImageIcon className="w-3.5 h-3.5" /> {p._count.images}
   434→                        </button>
   435→                      </td>
   436→                      <td className="py-2.5 px-3 text-center">
   437→                        <Switch checked={p.featured} onCheckedChange={() => toggleFeatured(p)} className="mx-auto" />
   438→                      </td>
   439→                      <td className="py-2.5 px-3 text-center">
   440→                        <Switch checked={isVisible} onCheckedChange={() => toggleVisible(p)} className="mx-auto" />
   441→                      </td>
   442→                      <td className="py-2.5 px-3 text-center">
   443→                        <div className="flex items-center justify-center gap-1">
   444→                          <button onClick={() => openEdit(p)} className="p-1.5 rounded-md hover:bg-blue-50 text-blue-600 transition-colors">
   445→                            <Pencil className="w-4 h-4" />
   446→                          </button>
   447→                          <button onClick={() => setDeleteTarget(p)} className="p-1.5 rounded-md hover:bg-red-50 text-red-500 transition-colors">
   448→                            <Trash2 className="w-4 h-4" />
   449→                          </button>
   450→                        </div>
   451→                      </td>
   452→                    </tr>
   453→                    );
   454→                  })}
   455→                </tbody>
   456→              </table>
   457→            </div>
   458→          )}
   459→        </CardContent>
   460→      </Card>
   461→
   462→      {/* Pagination */}
   463→      {totalPages > 1 && (
   464→        <div className="flex items-center justify-center gap-2">
   465→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(p => p - 1)}>
   466→            <ChevronRight className="w-4 h-4" />
   467→          </Button>
   468→          <span className="text-sm text-muted-foreground">
