   203→const formatPrice = (price: number | null) => {
   204→  if (price === null) return 'استعلام قیمت';

   208→const useDebounce = (value: string, delay: number) => {
   209→  const [debounced, setDebounced] = useState(value);
   210→  useEffect(() => {
   211→    const t = setTimeout(() => setDebounced(value), delay);
   212→    return () => clearTimeout(t);
   213→  }, [value, delay]);
   214→  return debounced;
   215→};

   262→function ProductCard({ product, onClick, user, onAuthRequired, onAddToCart, onToggleCompare, isInCompare }: { product: Product; onClick: () => void; user: UserData | null; onAuthRequired: () => void; onAddToCart: (productId: number) => void; onToggleCompare: () => void; isInCompare: boolean }) {
   263→  const av = availabilityLabel(product.availability);
   264→  const hasDiscount = product.discountPercent > 0 && product.price !== null && product.priceType !== 'call';
   265→  const discountedPrice = hasDiscount ? Math.round(product.price! * (1 - product.discountPercent / 100)) : null;
   266→  const canAdd = product.availability === 'in_stock' && product.price !== null && product.priceType !== 'call';
   267→  const [adding, setAdding] = useState(false);
   268→
   269→  const handleAddToCart = async (e: React.MouseEvent) => {
   270→    e.stopPropagation();
   271→    if (!user) { onAuthRequired(); return; }
   272→    if (adding) return;
   273→    setAdding(true);
   274→    await onAddToCart(product.id);
   275→    setAdding(false);
   276→  };
   277→
   278→  return (
   279→    <motion.div whileHover={{ y: -4 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   280→      <Card
   281→        className="cursor-pointer group h-full border border-gray-100 hover:border-mersadent-accent/40 hover:shadow-lg transition-all duration-300 overflow-hidden"
   282→        onClick={onClick}
   283→      >
   284→        <CardContent className="p-3 flex flex-col gap-2">
   285→          {/* Image or placeholder */}
   286→          <div className="relative w-full aspect-square bg-gradient-to-br from-gray-50 to-gray-100 rounded-lg flex items-center justify-center group-hover:from-mersadent-accent/5 group-hover:to-mersadent-accent/10 transition-colors overflow-hidden">
   287→            {product.images && product.images.length > 0 ? (
   288→              <img
   289→                src={product.images[0].url}
   290→                alt={product.images[0].alt || product.name}
   291→                className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
   292→                loading="lazy"
   293→              />
   294→            ) : (
   295→              <Package className="w-10 h-10 text-gray-300 group-hover:text-mersadent-accent/50 transition-colors" />
   296→            )}
   297→            {/* Top badges: featured + discount */}
   298→            <div className="absolute top-1.5 left-1.5 flex flex-col gap-1">
   299→              {hasDiscount && (
   300→                <Badge className="bg-red-500 text-white text-[9px] px-1.5 py-0">
   301→                  تخفیف {product.discountPercent}%
   302→                </Badge>
   303→              )}
   304→              {product.featured && (
   305→                <Badge className="bg-amber-500 text-white text-[9px] px-1 py-0">
   306→                  <Star className="w-2.5 h-2.5 ml-0.5" />
   307→                  ویژه
   308→                </Badge>
   309→              )}
   310→            </div>
   311→            {/* Compare toggle button */}
   312→            <button
   313→              onClick={(e) => { e.stopPropagation(); onToggleCompare(); }}
   314→              className={`absolute bottom-1.5 left-1.5 w-7 h-7 rounded-full flex items-center justify-center transition-all duration-200 z-10 ${isInCompare ? 'bg-teal-500 text-white shadow-md opacity-100' : 'bg-white/90 text-gray-400 hover:text-teal-500 hover:bg-white shadow-sm opacity-80 group-hover:opacity-100'}`}
   315→              title={isInCompare ? 'حذف از لیست مقایسه' : 'افزودن به لیست مقایسه'}
   316→              aria-label={isInCompare ? 'حذف از مقایسه' : 'افزودن به مقایسه'}
   317→              aria-pressed={isInCompare}
   318→            >
   319→              <GitCompareArrows className="w-3.5 h-3.5" />
   320→            </button>
   321→            {/* Add to cart button */}
   322→            {canAdd && (
   323→              <button
   324→                onClick={handleAddToCart}
   325→                disabled={adding}
   326→                className="absolute bottom-1.5 left-9 w-7 h-7 bg-white/90 hover:bg-mersadent-accent text-mersadent-dark hover:text-white rounded-full flex items-center justify-center shadow-md opacity-0 group-hover:opacity-100 transition-all duration-200 disabled:opacity-60"
   327→                title="افزودن به سبد خرید"
   328→              >
   329→                {adding ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <ShoppingCart className="w-3.5 h-3.5" />}
   330→              </button>
   331→            )}
   332→            {product.availability === 'out_of_stock' && (
   333→              <div className="absolute inset-0 bg-white/70 rounded-lg flex items-center justify-center">
   334→                <Badge variant="secondary" className="text-red-500 text-xs">ناموجود</Badge>
   335→              </div>
   336→            )}
   337→            {/* Product tags */}
   338→            {product.tags && product.tags.length > 0 && (
   339→              <div className="absolute top-1.5 right-1.5 flex flex-col gap-1">
   340→                {product.tags.filter(t => t.tag.type === 'badge' || t.tag.type === 'consumable').map((t) => (
   341→                  <Badge
   342→                    key={t.tag.id}
   343→                    className={`text-[8px] px-1.5 py-0 font-medium ${
   344→                      t.tag.type === 'consumable'
   345→                        ? 'bg-orange-100 text-orange-700 border-orange-200'
   346→                        : 'bg-mersadent-accent/10 text-mersadent-accent border-mersadent-accent/20'
   347→                    }`}
   348→                  >
   349→                    {t.tag.name}
   350→                  </Badge>
   351→                ))}
   352→              </div>
   353→            )}
   354→          </div>
   355→
   356→          {/* Info */}
   357→          <div className="flex-1 flex flex-col gap-1 min-h-0">
   358→            <span className="text-[10px] text-gray-400 font-medium">{product.brand.name}</span>
   359→            <h3 className="text-xs font-semibold text-mersadent-dark leading-5 line-clamp-2">
   360→              {product.name}
   361→            </h3>
   362→          </div>
   363→
   364→          {/* Price & availability */}
   365→          <div className="flex flex-col gap-1 pt-1.5 border-t border-gray-100">
   366→            {product.priceType === 'call' || product.price === null ? (
   367→              <Badge className="w-fit bg-mersadent-accent/10 text-mersadent-accent border-mersadent-accent/20 text-xs font-medium">
   368→                تماس بگیرید
   369→              </Badge>
   370→            ) : hasDiscount && discountedPrice !== null ? (
   371→              <div className="flex flex-col gap-0.5">
   372→                <span className="text-[11px] text-gray-400 line-through">
   373→                  {formatPrice(product.price)}
   374→                </span>
   375→                <span className="text-sm font-bold text-red-600">
   376→                  {formatPrice(discountedPrice)}
   377→                </span>
   378→              </div>
   379→            ) : (
   380→              <span className="text-sm font-bold text-mersadent-dark">
   381→                {formatPrice(product.price)}
   382→              </span>
   383→            )}
   384→            <span className={`text-[11px] px-2 py-0.5 rounded-full w-fit ${av.color}`}>
   385→              {av.text}
   386→            </span>
   387→          </div>
   388→        </CardContent>
   389→      </Card>
   390→    </motion.div>
   391→  );
   392→}

   426→  const inputRef = useRef<HTMLInputElement>(null);

   431→      queueMicrotask(() => {
   432→        setSearchResults([]);
   433→        setSearchOpen(false);
   434→      });
   435→      return;
   436→    }
   437→    fetch(`/api/search?q=${encodeURIComponent(debounced)}`)

   439→      .then(d => {
   440→        setSearchResults(d.results || []);

   465→      setSearchOpen(false);
   466→      onSearch(searchQuery.trim());

   473→    { label: 'درباره ما', action: () => { setMobileOpen(false); onNavigate({ type: 'about' }); } },
   474→  ];
   475→
   476→  // Mega menu hover handlers with delay
   477→  const handleCatEnter = (cat: Category | null) => {
   478→    if (catTimeoutRef.current) clearTimeout(catTimeoutRef.current);
   479→    setActiveCat(cat);
   480→  };
   481→  const handleCatLeave = () => {
   482→    catTimeoutRef.current = setTimeout(() => {
   483→      setActiveCat(null);
   484→      setCatMenuOpen(false);
   485→    }, 200);
   486→  };
   487→
   488→  return (
   489→    <header className="sticky top-0 z-50 bg-white border-b border-gray-200/80 shadow-sm">
   490→      <div className="max-w-7xl mx-auto px-4 sm:px-6">
   491→        <div className="flex items-center justify-between h-14 gap-4">
   492→          {/* Logo + Brand Name */}
   493→          <button
   494→            onClick={() => onNavigate({ type: 'home' })}
   495→            className="flex-shrink-0 flex items-center gap-2.5 cursor-pointer group"
   496→          >
   497→            <img src="/logo.png" alt="MersaDent" className="h-8 sm:h-9 w-auto object-contain" />
   498→            <div className="flex flex-col leading-none">
   499→              <span className="text-xl sm:text-2xl font-extrabold tracking-tight text-mersadent-dark group-hover:text-mersadent-accent transition-colors" style={{ fontFamily: "'Geist Mono', monospace" }}>MersaDent</span>
   500→              <span className="text-[11px] sm:text-xs font-bold text-mersadent-accent/70 mt-0.5">مرصادنت</span>
   501→            </div>
   502→          </button>
   503→
   504→          {/* Desktop Nav with Mega Menu */}
   505→          <nav className="hidden md:flex items-center gap-1">
   506→            {/* Categories Mega Menu */}
   507→            <div
   508→              ref={megaMenuRef}
   509→              className="relative"
   510→              onMouseEnter={() => { setCatMenuOpen(true); if (catTimeoutRef.current) clearTimeout(catTimeoutRef.current); }}
   511→              onMouseLeave={handleCatLeave}
   512→            >
   513→              <button
   514→                className={`flex items-center gap-1.5 text-sm font-medium px-3 py-2 rounded-lg transition-colors ${catMenuOpen ? 'text-mersadent-accent bg-mersadent-accent/5' : 'text-gray-600 hover:text-mersadent-accent'}`}
   515→              >
   516→                دسته‌بندی‌ها
   517→                <ChevronDown className={`w-3.5 h-3.5 transition-transform ${catMenuOpen ? 'rotate-180' : ''}`} />
   518→              </button>
   519→
   520→              {/* Mega Menu Dropdown */}
   521→              <AnimatePresence>
   522→                {catMenuOpen && (
   523→                  <motion.div
   524→                    initial={{ opacity: 0, y: 8 }}
   525→                    animate={{ opacity: 1, y: 0 }}

   858→// Resolve a hero slide's CTA link into a navigation action.
   859→// - Internal app paths like "/category/all" → in-app navigation
   860→// - Empty link → default "all categories" view
   861→// - External URLs (http/https) → open in new tab via window.open
   862→function resolveHeroLink(link: string | undefined, onNavigate: (view: PageView) => void) {
   863→  const trimmed = (link || '').trim();
   864→  if (!trimmed) {
   865→    onNavigate({ type: 'category', categoryId: 0, slug: 'all' });
   866→    return;
   867→  }
   868→  if (/^https?:\/\//i.test(trimmed)) {
   869→    window.open(trimmed, '_blank', 'noopener,noreferrer');
   870→    return;
   871→  }
   872→  // Internal path: try to parse common patterns
   873→  // /category/all, /category/<id>, /category/<slug>, /brand/all, /about, /search?q=...
   874→  const catMatch = trimmed.match(/^\/category\/(.+)$/);
   875→  if (catMatch) {
   876→    const rest = catMatch[1];
   877→    if (rest === 'all') {
   878→      onNavigate({ type: 'category', categoryId: 0, slug: 'all' });
   879→    } else if (/^\d+$/.test(rest)) {
   880→      onNavigate({ type: 'category', categoryId: parseInt(rest), slug: rest });
   881→    } else {
   882→      onNavigate({ type: 'category', categoryId: 0, slug: rest });
   883→    }
   884→    return;
   885→  }
   886→  if (/^\/brand(\/|$)/.test(trimmed)) {
   887→    onNavigate({ type: 'brand', brandId: 0, brandName: 'all' });
   888→    return;
   889→  }
   890→  if (/^\/about/.test(trimmed)) {
   891→    onNavigate({ type: 'about' });
   892→    return;
   893→  }
   894→  if (/^\/blog/.test(trimmed)) {
   895→    onNavigate({ type: 'blog' });
   896→    return;
   897→  }
   898→  if (/^\/search\?/.test(trimmed)) {
   899→    const q = new URLSearchParams(trimmed.split('?')[1]).get('q') || '';
   900→    onNavigate({ type: 'search', query: q });
   901→    return;
   902→  }
   903→  // Fallback: open as external/new-tab
   904→  window.open(trimmed, '_blank', 'noopener,noreferrer');
   905→}

   907→function HeroSlider({ onNavigate }: { onNavigate: (view: PageView) => void }) {
   908→  const [current, setCurrent] = useState(0);
   909→  const [isTransitioning, setIsTransitioning] = useState(false);
   910→  const [heroSlides, setHeroSlides] = useState<HeroSlide[]>(DEFAULT_HERO_SLIDES);
   911→  const [loaded, setLoaded] = useState(false);
   912→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   913→  const touchStartX = useRef(0);
   914→  const currentRef = useRef(0);
   915→
   916→  // Fetch admin-configured slides
   917→  useEffect(() => {
   918→    let cancelled = false;
   919→    fetch('/api/hero')
   920→      .then(r => r.json())
   921→      .then(d => {
   922→        if (cancelled) return;
   923→        if (Array.isArray(d.slides) && d.slides.length > 0) {
   924→          setHeroSlides(d.slides.map((s: HeroSlide) => ({ ...s, objectPosition: 'center' })));
   925→        }
   926→      })
   927→      .catch(() => {})
   928→      .finally(() => { if (!cancelled) setLoaded(true); });
   929→    return () => { cancelled = true; };
   930→  }, []);
   931→
   932→  const slidesCount = heroSlides.length;
   933→
   934→  const goTo = useCallback((index: number) => {
   935→    if (isTransitioning) return;
   936→    setIsTransitioning(true);
   937→    setCurrent(index);
   938→    currentRef.current = index;
   939→    setTimeout(() => setIsTransitioning(false), 700);
   940→  }, [isTransitioning]);
   941→
   942→  const next = useCallback(() => goTo((current + 1) % slidesCount), [current, goTo, slidesCount]);
   943→  const prev = useCallback(() => goTo((current - 1 + slidesCount) % slidesCount), [current, goTo, slidesCount]);
   944→
   945→  // Auto-play
   946→  useEffect(() => {
   947→    if (!loaded || slidesCount <= 1) return;
   948→    intervalRef.current = setInterval(() => {
   949→      goTo((currentRef.current + 1) % slidesCount);
   950→    }, 6000);
   951→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   952→  }, [goTo, slidesCount, loaded]);
   953→
   954→  // Pause on hover
   955→  const pause = () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   956→  const resume = () => {
   957→    if (slidesCount <= 1) return;
   958→    intervalRef.current = setInterval(() => {
   959→      goTo((currentRef.current + 1) % slidesCount);
   960→    }, 6000);
   961→  };
   962→
   963→  // Touch handlers for swipe
   964→  const onTouchStart = (e: React.TouchEvent) => { touchStartX.current = e.touches[0].clientX; };
   965→  const onTouchEnd = (e: React.TouchEvent) => {
   966→    const diff = touchStartX.current - e.changedTouches[0].clientX;
   967→    if (Math.abs(diff) > 50) { if (diff > 0) next(); else prev(); }
   968→  };
   969→
   970→  const slide = heroSlides[current] || heroSlides[0];
   971→
